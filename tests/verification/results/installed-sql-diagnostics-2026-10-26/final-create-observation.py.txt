import ctypes, hashlib, json, sys, time, urllib.request
from pathlib import Path

source = Path('/var/tmp/transit-ci-8dmwlxgr/src')
sys.path.insert(0, str(source / 'tests'))
from server import Fixture

assert ctypes.CDLL(None).prctl(36, 1, 0, 0, 0) == 0
root = Path('/var/tmp/transit-server-proof-budxzd8h')
f = Fixture.__new__(Fixture)
f.root = root
f.cli = str(root / 'transit')
f.path = root / 'transit.yaml'
f.config = json.loads(f.path.read_text())
f.no_ui = True
f.process = None
f.logs = [p.open('rb') for p in sorted(root.glob('manager-*.stderr'))]
f.activation = None
f.markers = set()
f.reserved_ports = set()
f.http = urllib.request.build_opener(urllib.request.ProxyHandler({}))
f.url = 'http://127.0.0.1:' + str(f.config['ports']['gateway'])
record = dict(installation=str(root), revision='e3dd9e68a5bb777169bbbb4835ee201f7f0a77d8',
              started=time.time(), complete=False,
              actions=['restart the same retained installation',
                       'read public original operation inventory and Table names',
                       'do not resubmit the original SQL'])
record['artifacts'] = {name: dict(path=str(path), sha256=hashlib.sha256(Path(path).read_bytes()).hexdigest())
                       for name, path in dict(cli=f.cli, **f.config['executables']).items()}
destination = Path('/var/tmp/transit-lifecycle-final-create-observation.json')
destination.write_text(json.dumps(record, indent=2) + '\n')
try:
    f.start()
    f.ready(timeout=300)
    record['operations'] = f.sql_operations()
    record['tables'] = f.console('SHOW TABLES;\n', request_timeout_ms=120000, timeout=180)
    record['complete'] = True
except BaseException as error:
    record['failure'] = repr(error)
    raise
finally:
    f.cleanup()
    record['finished'] = time.time()
    destination.write_text(json.dumps(record, indent=2) + '\n')
    print(json.dumps(record), flush=True)
