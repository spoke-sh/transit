import hashlib, json, os, sys, time
from pathlib import Path

source = Path('/var/tmp/transit-ci-8dmwlxgr/src')
sys.path.insert(0, str(source / 'tests'))
import server_direct_tables as proof

base = Path('/var/tmp/transit-lifecycle-current-installed-diagnostic')
record = dict(revision='e3dd9e68a5bb777169bbbb4835ee201f7f0a77d8',
              started=time.time(), complete=False,
              method='Exact source and binaries; exec wrapper captures native stderr. Run after the heavy repository unit stage finishes. No change to workload, deadlines, authority or recovery assertions.')
artifacts = [source / '.zig-cache/o/b9bcd49f167ac364009580cc3c7540ee/transit',
             source / '.zig-cache/o/2f63109d261f7c735684cdbff217113b/transit-node',
             source / '.zig-cache/o/41755c653e04095696c17101e33a6137/transit-query-host',
             source / '.zig-cache/o/1ca65599c4afe83c06dc4e151b834336/transit-board',
             source / '.zig-cache/o/d0e9d0dfc306ebf63a5b01f5384d40e4/transit-sql-gateway']

original = proof.LifecycleFixture.__init__

def traced(self, *args, **kwargs):
    original(self, *args, **kwargs)
    node = self.config['executables']['node']
    wrapper = self.root / 'node-diagnostics'
    capture = self.root / 'native.stderr'
    code = ('#!' + sys.executable + '\nimport os,sys\nfd=os.open(' + repr(str(capture))
            + ',os.O_WRONLY|os.O_CREAT|os.O_APPEND,0o600)\nos.dup2(fd,2)\nos.close(fd)\nos.execv('
            + repr(node) + ',[' + repr(node) + ',*sys.argv[1:]])\n')
    wrapper.write_text(code)
    wrapper.chmod(0o700)
    self.config['executables']['node'] = str(wrapper)
    self.path.write_text(json.dumps(self.config))
    record['installation'] = str(self.root)
    record['native_stderr'] = str(capture)
    base.with_suffix('.json').write_text(json.dumps(record, indent=2) + '\n')
    print('Native diagnostic capture:', capture, flush=True)

proof.LifecycleFixture.__init__ = traced
try:
    proof.run(*map(str, artifacts))
    record['complete'] = True
except BaseException as error:
    record['failure'] = repr(error)
    raise
finally:
    record['finished'] = time.time()
    base.with_suffix('.json').write_text(json.dumps(record, indent=2) + '\n')
    print(json.dumps({k: record[k] for k in ('revision', 'complete', 'started', 'finished')}), flush=True)
